% \cvheader{photo}{name}{headline}{contact line}
\newcommand{\cvheader}[4]{%
  \begin{minipage}[c]{2.55cm}
    \begin{tikzpicture}
      \clip (0,0) circle (1.2cm);
      \node at (0,0) {\includegraphics[width=2.4cm]{#1}};
    \end{tikzpicture}
  \end{minipage}%
  \hspace{0.5cm}%
  \begin{minipage}[c]{\dimexpr\linewidth-3.05cm}
    {\fontsize{26}{28}\selectfont\bfseries #2}\\[3pt]
    {\large\color{accent} #3}\\[6pt]
    {\small #4}
  \end{minipage}
  \vspace{0.2cm}
}

% \job{role}{company}{place}{dates}
\newcommand{\job}[4]{%

}

% Small grey line with the technologies of a job or project.

}
